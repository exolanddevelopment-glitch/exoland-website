/**
 * Admin Sewa - payment web app for the Property DB.
 * Paste this into Extensions > Apps Script (file "Code.gs").
 *
 * Menu 1 "Catat Pembayaran": records a payment as a NEW row in DBRiwayatSewa (never overwrites).
 * Menu 2 "Tambah Bukti":     attaches proof (photo/PDF) to an existing payment later.
 *                             Only the Bukti cell of that payment is extended; nothing else changes.
 * DBStatusSewa gets 3 status columns (O, P, Q) via menu "Pembayaran > Pasang kolom status".
 */

var SHEET_TENANT = 'DBStatusSewa';
var SHEET_LOG = 'DBRiwayatSewa';
var PROOF_FOLDER = 'Bukti Pembayaran';

// DBStatusSewa columns (1-based), as in the current file
var T = { INV: 3, UNIT: 4, NAMA: 5, WA: 6, MULAI: 7, SELESAI: 8, SIKLUS: 9, HARGA: 10, STATUS: 12, TEMPO: 13 };

var LOG_HEADERS = ['ID Bayar', 'Waktu Input', 'Tanggal Bayar', 'Inventaris', 'No Unit', 'Nama Penyewa',
  'Siklus Bayar', 'Harga Sewa', 'Jumlah Periode', 'Jumlah Bayar', 'Periode Mulai', 'Lunas s/d',
  'Metode', 'Bukti', 'Catatan', 'Dicatat Oleh', 'Bukti Diupdate', 'Bukti Diupdate Oleh'];
var L = { ID: 1, INPUT: 2, TGL: 3, INV: 4, UNIT: 5, NAMA: 6, SIKLUS: 7, HARGA: 8, N: 9, JUMLAH: 10,
  MULAI: 11, SD: 12, METODE: 13, BUKTI: 14, CATATAN: 15, OLEH: 16, BUKTI_UPD: 17, BUKTI_OLEH: 18 };

var STATUS_HEADERS = ['Lunas s/d', 'Bayar Terakhir', 'Status Bayar'];
var STATUS_COL = 15; // column O

/* ---------- menu inside the spreadsheet ---------- */
function onOpen() {
  SpreadsheetApp.getUi().createMenu('Pembayaran')
    .addItem('Pasang kolom status', 'setup')
    .addToUi();
}

/** Setup: log headers + status columns. Safe to run again. */
function setup() {
  var ss = SpreadsheetApp.getActive();
  ss.setSpreadsheetTimeZone(Session.getScriptTimeZone());
  var log = ss.getSheetByName(SHEET_LOG) || ss.insertSheet(SHEET_LOG);
  ensureLogHeaders_(log);

  var t = ss.getSheetByName(SHEET_TENANT);
  var head = t.getRange(1, STATUS_COL, 1, 3).getValues()[0];
  if (head.join('') !== '' && head[0] !== STATUS_HEADERS[0]) {
    throw new Error('Kolom O-Q di DBStatusSewa sudah terisi. Kosongkan dulu atau hubungi admin.');
  }
  t.getRange(1, STATUS_COL, 1, 3).setValues([STATUS_HEADERS]).setFontWeight('bold');
  var last = t.getLastRow();
  var f = [];
  for (var r = 2; r <= last; r++) {
    var key = 'DBRiwayatSewa!D:D,C' + r + ',DBRiwayatSewa!E:E,D' + r + ',DBRiwayatSewa!F:F,E' + r;
    f.push([
      '=IF(E' + r + '="","",IFERROR(1/(1/MAXIFS(DBRiwayatSewa!L:L,' + key + ')),""))',
      '=IF(E' + r + '="","",IFERROR(1/(1/MAXIFS(DBRiwayatSewa!C:C,' + key + ')),""))',
      '=IF(E' + r + '="","",IF(L' + r + '<>"Aktif","-",IF(O' + r + '="","Belum ada data",' +
        'IF(O' + r + '<TODAY(),"Telat",IF(O' + r + '-TODAY()<=7,"Segera jatuh tempo","Lunas")))))'
    ]);
  }
  if (f.length) {
    t.getRange(2, STATUS_COL, f.length, 3).setFormulas(f);
    t.getRange(2, STATUS_COL, f.length, 2).setNumberFormat('dd/mm/yyyy');
  }
  ss.toast('Kolom status terpasang di DBStatusSewa (O-Q).', 'Pembayaran', 5);
}

/** Writes any missing header in DBRiwayatSewa row 1 (also adds the 2 new Bukti columns to an old sheet). */
function ensureLogHeaders_(log) {
  var cur = log.getRange(1, 1, 1, LOG_HEADERS.length).getValues()[0];
  var fresh = cur.join('') === '';
  for (var i = 0; i < LOG_HEADERS.length; i++) {
    if (cur[i] === '') log.getRange(1, i + 1).setValue(LOG_HEADERS[i]).setFontWeight('bold');
  }
  if (fresh) {
    log.setFrozenRows(1);
    log.getRange('B:B').setNumberFormat('dd/mm/yyyy hh:mm');
    log.getRange('C:C').setNumberFormat('dd/mm/yyyy');
    log.getRange('K:L').setNumberFormat('dd/mm/yyyy');
    log.getRange('H:H').setNumberFormat('#,##0');
    log.getRange('J:J').setNumberFormat('#,##0');
  }
}

/* ---------- web app ---------- */
function doGet() {
  return HtmlService.createHtmlOutputFromFile('Form')
    .setTitle('Admin Sewa')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function tz_() { return Session.getScriptTimeZone(); }
function me_() { return Session.getActiveUser().getEmail() || ''; }

function cycleMonths_(siklus) {
  var s = String(siklus || '').toLowerCase();
  if (s.indexOf('tahun') >= 0) return 12;
  if (s.indexOf('6') >= 0) return 6;
  if (s.indexOf('3') >= 0) return 3;
  return 1;
}

/** Add k months, keeping the day (31 Jan + 1 month = 28/29 Feb). Result at 12:00. */
function addMonths_(d, k) {
  var y = d.getFullYear(), m = d.getMonth() + k;
  var last = new Date(y, m + 1, 0).getDate();
  return new Date(y, m, Math.min(d.getDate(), last), 12);
}

function keyOf_(inv, unit, nama) {
  return [String(inv), String(unit), String(nama).trim()].join('|');
}

function fmt_(d) {
  return d instanceof Date ? Utilities.formatDate(d, tz_(), 'dd/MM/yyyy') : String(d || '');
}
function iso_(d) {
  return d instanceof Date ? Utilities.formatDate(d, tz_(), 'yyyy-MM-dd') : '';
}

/** Active tenants with the exact cell values (server side only). */
function loadTenants_() {
  var t = SpreadsheetApp.getActive().getSheetByName(SHEET_TENANT);
  var rows = t.getRange(2, 1, Math.max(t.getLastRow() - 1, 1), 13).getValues();
  var tenants = [];
  rows.forEach(function (r) {
    var nama = String(r[T.NAMA - 1] || '').trim();
    if (!nama) return;
    if (String(r[T.STATUS - 1]).trim() !== 'Aktif') return;
    tenants.push({
      key: keyOf_(r[T.INV - 1], r[T.UNIT - 1], nama),
      inv: String(r[T.INV - 1] || ''),
      unit: String(r[T.UNIT - 1] || ''),
      nama: nama,
      siklus: String(r[T.SIKLUS - 1] || 'Bulanan'),
      harga: Number(r[T.HARGA - 1]) || 0,
      raw: [r[T.INV - 1], r[T.UNIT - 1], r[T.NAMA - 1]]  // exact cell values, so the status formulas match
    });
  });
  tenants.sort(function (a, b) { return a.nama.localeCompare(b.nama); });
  return tenants;
}

/** Data the app needs when it opens. */
function getFormData() {
  var list = loadTenants_().map(function (x) {
    return { key: x.key, inv: x.inv, unit: x.unit, nama: x.nama, siklus: x.siklus, harga: x.harga };
  });
  return { tenants: list, user: me_() };
}

function logRows_() {
  var log = SpreadsheetApp.getActive().getSheetByName(SHEET_LOG);
  if (!log || log.getLastRow() < 2) return { log: log, rows: [] };
  return { log: log, rows: log.getRange(2, 1, log.getLastRow() - 1, LOG_HEADERS.length).getValues() };
}

function proofLinks_(cell) {
  return String(cell || '').split(/\s+/).filter(function (x) { return /^https?:\/\//.test(x); });
}

/** Last payments of one tenant, to show before saving and to catch duplicates. */
function getHistory(key) {
  var out = [];
  logRows_().rows.forEach(function (r) {
    if (keyOf_(r[L.INV - 1], r[L.UNIT - 1], r[L.NAMA - 1]) === key) {
      out.push({ tgl: fmt_(r[L.TGL - 1]), jumlah: Number(r[L.JUMLAH - 1]) || 0,
        mulai: fmt_(r[L.MULAI - 1]), sd: fmt_(r[L.SD - 1]), mulaiIso: iso_(r[L.MULAI - 1]), sdIso: iso_(r[L.SD - 1]) });
    }
  });
  out.sort(function (a, b) { return a.sdIso < b.sdIso ? 1 : -1; });  // latest period first
  return out.slice(0, 5);
}

function paymentObj_(r) {
  return {
    id: String(r[L.ID - 1]),
    input: r[L.INPUT - 1] instanceof Date ? Utilities.formatDate(r[L.INPUT - 1], tz_(), 'dd/MM/yyyy HH:mm') : '',
    inputIso: r[L.INPUT - 1] instanceof Date ? r[L.INPUT - 1].toISOString() : '',
    tgl: fmt_(r[L.TGL - 1]),
    inv: String(r[L.INV - 1]), unit: String(r[L.UNIT - 1]), nama: String(r[L.NAMA - 1]).trim(),
    siklus: String(r[L.SIKLUS - 1]), n: Number(r[L.N - 1]) || 1, jumlah: Number(r[L.JUMLAH - 1]) || 0,
    mulai: fmt_(r[L.MULAI - 1]), sd: fmt_(r[L.SD - 1]),
    metode: String(r[L.METODE - 1]), catatan: String(r[L.CATATAN - 1]), oleh: String(r[L.OLEH - 1]),
    bukti: proofLinks_(r[L.BUKTI - 1]),
    buktiUpd: r[L.BUKTI_UPD - 1] instanceof Date ? Utilities.formatDate(r[L.BUKTI_UPD - 1], tz_(), 'dd/MM/yyyy HH:mm') : '',
    buktiOleh: String(r[L.BUKTI_OLEH - 1] || '')
  };
}

/** All logged payments, newest input first (max 1000), for menu "Tambah Bukti". */
function getPayments() {
  var list = [];
  logRows_().rows.forEach(function (r) { if (r[L.ID - 1]) list.push(paymentObj_(r)); });
  list.sort(function (a, b) { return a.inputIso < b.inputIso ? 1 : -1; });
  return list.slice(0, 1000);
}

/** Saves uploaded files to Drive folder "Bukti Pembayaran" and returns their links. */
function saveFiles_(files, baseName) {
  if (!files || !files.length) return [];
  var folders = DriveApp.getFoldersByName(PROOF_FOLDER);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(PROOF_FOLDER);
  var stamp = Utilities.formatDate(new Date(), tz_(), 'yyyyMMdd-HHmmss');
  return files.map(function (f, i) {
    var mime = /^(image\/jpeg|application\/pdf)$/.test(f.mime) ? f.mime : 'image/jpeg';
    var ext = mime === 'application/pdf' ? '.pdf' : '.jpg';
    var blob = Utilities.newBlob(Utilities.base64Decode(f.data), mime, baseName + ' ' + stamp + '-' + (i + 1) + ext);
    return folder.createFile(blob).getUrl();
  });
}

/** Save one payment. Validates again on the server. Proof is optional (can be added later). */
function savePayment(p) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var ten = null;
    loadTenants_().forEach(function (x) { if (x.key === p.key) ten = x; });
    if (!ten) throw new Error('Penyewa tidak ditemukan atau kontrak tidak aktif.');

    var jumlah = Number(p.jumlah);
    if (!(jumlah > 0)) throw new Error('Jumlah bayar harus angka lebih dari 0.');
    var n = Math.max(1, Math.round(Number(p.periode) || 1));
    if (!/^\d{4}-\d{2}-\d{2}$/.test(p.tanggal)) throw new Error('Tanggal bayar tidak valid.');
    if (!/^\d{4}-\d{2}-\d{2}$/.test(p.mulai)) throw new Error('Periode mulai tidak valid.');

    // Dates are saved at 12:00 noon so a time-zone difference can never shift them by a day.
    var dp = p.tanggal.split('-');
    var tgl = new Date(Number(dp[0]), Number(dp[1]) - 1, Number(dp[2]), 12);
    var today = new Date(); today.setHours(0, 0, 0, 0);
    if (tgl > new Date(today.getTime() + 86400000)) throw new Error('Tanggal bayar tidak boleh di masa depan.');

    var parts = p.mulai.split('-');
    var mulai = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]), 12);
    var sd = addMonths_(mulai, cycleMonths_(ten.siklus) * n);
    if (sd.getDate() === mulai.getDate()) sd.setDate(sd.getDate() - 1);  // 22/10 + 2 months = 21/12; 31/01 + 1 month = 28/02

    var who = me_() || String(p.petugas || '').trim();
    if (!who) throw new Error('Isi nama petugas.');

    var links = saveFiles_(p.files, Utilities.formatDate(tgl, tz_(), 'yyyyMMdd') + ' ' + ten.inv + ' ' + ten.unit + ' ' + ten.nama);

    var log = SpreadsheetApp.getActive().getSheetByName(SHEET_LOG);
    ensureLogHeaders_(log);
    var id = 'PAY-' + Utilities.formatDate(new Date(), tz_(), 'yyyyMMdd-HHmmss') + '-' + Math.floor(Math.random() * 900 + 100);
    var now = new Date();
    log.appendRow([id, now, tgl, ten.raw[0], ten.raw[1], ten.raw[2], ten.siklus, ten.harga, n, jumlah,
      mulai, sd, String(p.metode || ''), links.join('\n'), String(p.catatan || '').slice(0, 500), who,
      links.length ? now : '', links.length ? who : '']);
    var r = log.getLastRow();
    log.getRange(r, L.INPUT).setNumberFormat('dd/mm/yyyy hh:mm');
    log.getRange(r, L.TGL).setNumberFormat('dd/mm/yyyy');
    log.getRange(r, L.MULAI, 1, 2).setNumberFormat('dd/mm/yyyy');
    log.getRange(r, L.HARGA).setNumberFormat('#,##0');
    log.getRange(r, L.JUMLAH).setNumberFormat('#,##0');
    log.getRange(r, L.BUKTI_UPD).setNumberFormat('dd/mm/yyyy hh:mm');
    return { id: id, sd: fmt_(sd), nama: ten.nama, bukti: links.length };
  } finally {
    lock.releaseLock();
  }
}

/** Menu "Tambah Bukti": add one or more proof files to an existing payment. Existing proof is kept. */
function addProof(p) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    if (!p || !p.id) throw new Error('Pilih pembayaran dulu.');
    if (!p.files || !p.files.length) throw new Error('Pilih minimal 1 file bukti.');
    var who = me_() || String(p.petugas || '').trim();
    if (!who) throw new Error('Isi nama petugas.');

    var data = logRows_();
    var idx = -1;
    data.rows.forEach(function (r, i) { if (String(r[L.ID - 1]) === String(p.id)) idx = i; });
    if (idx < 0) throw new Error('Pembayaran ' + p.id + ' tidak ditemukan.');
    var row = data.rows[idx], sheetRow = idx + 2;

    var links = saveFiles_(p.files, fmt_(row[L.TGL - 1]).replace(/\//g, '') + ' ' + row[L.INV - 1] + ' ' + row[L.UNIT - 1] + ' ' + String(row[L.NAMA - 1]).trim());
    var all = proofLinks_(row[L.BUKTI - 1]).concat(links);
    var log = data.log;
    ensureLogHeaders_(log);
    log.getRange(sheetRow, L.BUKTI).setValue(all.join('\n'));
    log.getRange(sheetRow, L.BUKTI_UPD).setValue(new Date()).setNumberFormat('dd/mm/yyyy hh:mm');
    log.getRange(sheetRow, L.BUKTI_OLEH).setValue(who);
    SpreadsheetApp.flush();
    return paymentObj_(log.getRange(sheetRow, 1, 1, LOG_HEADERS.length).getValues()[0]);
  } finally {
    lock.releaseLock();
  }
}
